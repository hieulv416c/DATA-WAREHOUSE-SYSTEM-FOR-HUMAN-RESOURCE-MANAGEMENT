    [Dim Emplyee].[Employee Key].members ON ROWS
FROM [DTWH FINAL]
-- Bài 12: Tính lương 1 giờ của mỗi nhân viên 
--để phục vụ cho việc tính toán lương 1 tháng và lương làm thêm của nhân viên
SELECT 
    [Measures].[SalaryMonth] ON COLUMNS,
    [Dim Emplyee].[Employee Key].members ON ROWS
FROM [DTWH FINAL]
-- Bài 13: Tạo AverageHoursPerEmployeePerDay tính toán xem nhân viên nào tích cưc làm việc nhất
-- với nhiều giờ làm việc nhất để khen thưởng.
WITH 
MEMBER [Measures].[AverageHoursPerEmployeePerDay] AS
    IIF(
        ([Measures].[Total Hours Worked] = 0 OR [Measures].[Fact Employee Salary Count] = 0),
        NULL, -- Tránh chia cho 0
        SUM(
            [Dim Time].[The Year].&[2010], 
            [Measures].[Total Hours Worked]
        ) / ([Measures].[Fact Employee Salary Count] * 365)
    )
SELECT 
    [Measures].[AverageHoursPerEmployeePerDay] ON COLUMNS,
     [Dim Emplyee].[Employee Key].members ON ROWS
FROM [DTWH FINAL]
